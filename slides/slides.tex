% === 5. The multi-index models ============================================
\begin{frame}{The multi-index models}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \role{\large Gaussian data}
      \vspace{-0.4em}
      \[ \vec{x} \sim \mathcal{N}(0, I_d) \]

      \role{\large The target function}
      \vspace{-0.4em}
      \[ f^\star(\vec{x}) = h^\star\!\left(\frac{W^\star \vec{x}}{\sqrt{d}}\right) \]
      \vspace{-1.0em}
      \begin{center}
        \small
        $h^\star : \mathbb{R}^k \to \mathbb{R}$,\qquad
        $W^\star = \{\vec{w}^\star_r\}_{r \in [k]} \in \mathbb{R}^{k \times d}$
        % \par\vspace{0.25em}
        % \aside{we call $f^\star$ the \emph{teacher}}\par
        % \aside{$k = \mathcal{O}(1) \ll d$;\;
        %   $k = 1$ is the \emph{single-index} model}
      \end{center}
      \vspace{0.2em}

      \role{\large (Possible) label noise}
      % \vspace{-0.4em}
      \[
        y = f^\star(\vec{x}) + \sqrt{\Delta}\,\xi,
        \qquad \xi \sim \mathcal{N}(0,1)
      \]
    \end{column}
    \begin{column}{0.50\textwidth}
      \centering
      \thesisnet[0.82\linewidth]{chapters/figs/multi-index.tex}
      \vspace{-1.0em}
      \begin{defbox}{Teacher subspace}
        \centering
        $V^\star = \operatorname{span}(\vec{w}^\star_1, \ldots, \vec{w}^\star_k)$,
        \qquad $\dim V^\star = k \ll d$
        \vspace{0.15em}

        \emph{Learning $\equiv$ discovering $V^\star$}
      \end{defbox}
    \end{column}
  \end{columns}
  \vspace{1.2em}
  \punchline{These $k$ directions \emph{are} the features of the problem.}
  \note{\notetime{5:29}\begin{notelist}
    \item Gaussian input, no correlation across components
    \item \textbf{$d$ is large for the whole talk; $k$ is fixed and does \emph{not}
      scale with $d$}
    \item target $f^\star$: a function of a low-dimensional projection
    \item label noise, to mimic real data
    \item rows of $W^\star$ = teacher subspace; \emph{these are the features}
    \item learning $\equiv$ discovering that subspace
    \item \textbf{why ``teacher'': it is the oracle that generates the data, and it
      teaches the student}
    \item \textbf{$\to$ next: and the student is the simplest net you can imagine}
  \end{notelist}}
\end{frame}

% === 6. The learned model: 2-layer network ================================
\begin{frame}{The learned model: 2-layer network}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \[
        f(\vec{x};\, W, \vec{a})
        = \frac{1}{p} \sum_{j=1}^p a_j\,
          \sigma\!\left(\frac{\langle \vec{w}_j, \vec{x}\rangle}{\sqrt{d}}\right)
      \]
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \thesisnet[0.95\linewidth]{chapters/figs/two-layer-net.tex}
    \end{column}
  \end{columns}
  \vspace{0.3em}
  \punchline{The student is a \emph{special case} of the teacher, and it is a \emph{two-layer NN}.}
  \vspace{0.5em}
  \pause
  \begin{columns}[t]
    \begin{column}{0.42\textwidth}
      \centering
      \role{\large Loss function}
      \[ \ell(\hat{y}, y) = \tfrac{1}{2}(\hat{y} - y)^2 \]
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering
      \role{\large Population risk}
      \[
        \mathcal{R}(W, \vec{a}) = \mathbb{E}_{(\vec{x},\, y)}\!\left[
          \tfrac{1}{2}\bigl(f(\vec{x};\, W, \vec{a}) - y\bigr)^2\right]
      \]
    \end{column}
  \end{columns}
  \note{\notetime{6:40}\begin{notelist}
    \item simplest net: two layers, $W$ first, $\vec{a}$ second, $\sigma$ the
      activation
    \item \textbf{a special case of the multi-index model: what was $h^\star$ is now
      just a sum over the neurons of the second layer}
    \item to learn $=$ align $W$ with $W^\star$
    \item \emph{pause}
    \item square loss --- the natural one for regression
    \item \textbf{we know the distribution exactly, so we can measure the
      \emph{population} risk}
    \item best measure there is, and it measures generalization: the expectation is
      over the whole distribution
    \item \textbf{but it cannot be used for training --- that is data leakage. It is
      a measuring instrument for the toy setting only}
    \item \textbf{$\to$ next: pillar three, the algorithm}
  \end{notelist}}
\end{frame}

% === 7. The algorithm: one-pass SGD =======================================
\begin{frame}{The algorithm: one-pass SGD}
  \begin{columns}[c]
    \begin{column}{0.26\textwidth}\raggedleft\role{\large Update}\end{column}
    \begin{column}{0.74\textwidth}
      \[
        \vec{w}_j^{\,\nu+1} = \vec{w}_j^{\,\nu}
          - \gamma\,\nabla_{\vec{w}_j}\,
            \ell\!\left(f(\vec{x}^\nu;\, W^\nu, \vec{a}),\, y^\nu\right),
        \qquad j \in [p],
      \]
    \end{column}
  \end{columns}
  \vspace{-0.3em}
  \begin{center}
    \aside{$\gamma > 0$ is the \emph{learning rate}; each step draws a
      \emph{fresh} sample, uses it once, and never reuses it}
  \end{center}
  \vspace{0.8em}
  \begin{columns}[c]
    \begin{column}{0.26\textwidth}\raggedleft\role{\large Initialization}\end{column}
    \begin{column}{0.74\textwidth}
      \[ \vec{w}_j^0 \sim \mathcal{N}(0, I_d) \]
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \begin{columns}[c]
    \begin{column}{0.26\textwidth}\raggedleft\role{\large Steps $=$ samples}\end{column}
    \begin{column}{0.74\textwidth}
      After $n$ steps the algorithm has consumed exactly $n$ samples: counting
      steps \emph{is} counting samples.
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \begin{columns}[c]
    \begin{column}{0.26\textwidth}\raggedleft\role{\large The key property}\end{column}
    \begin{column}{0.74\textwidth}
      \punchline{Independence of the steps}
    \end{column}
  \end{columns}
  \vfill
  \pause
  \begin{center}
    \aside{We actually often consider a spherically constrained dynamics for
      simplicity:}\quad
      
      \vspace{0.7em }
    $\displaystyle \vec{w}^{\nu+1} = \sqrt{d}\,
      \frac{\vec{w}^\nu - \gamma\nabla_{\vec{w}}\ell(f(\vec{x}^\nu), y^\nu)}
           {\bigl\lVert \vec{w}^\nu
             - \gamma\nabla_{\vec{w}}\ell(f(\vec{x}^\nu), y^\nu)\bigr\rVert}$
  \end{center}
  \note{\notetime{7:54}\begin{notelist}
    \item a fresh sample every step, one gradient, then discarded forever
    \item random Gaussian initialization
    \item steps $=$ samples
    \item key property: independence of the steps --- \textbf{this is the pillar the
      whole theory is built on}
    \item \textbf{caveat: we do try larger batches later, and I will say so}
    \item \emph{pause}
    \item spherical: the weights are forced onto the sphere, for convenience
    \item \textbf{legitimate because only the \emph{alignment} matters --- fixing the
      norm just removes variables to track}
    \item \textbf{$\to$ next: three pillars done. Now the trick that makes all of
      this tractable}
  \end{notelist}}
\end{frame}

% === 8. The reduced dynamic ===============================================
% Three bands, top to bottom: the projections both networks see, the order
% parameters that fix their joint law, the risk that follows.  The order
% parameters get the green definition box -- they are the object the rest of
% the talk is written in -- rather than being squeezed onto one \fiteq line
% with the covariance.
\begin{frame}{The reduced dynamic}
  \begin{columns}[c]
    \begin{column}{0.27\textwidth}
      \raggedleft
      \role{\large Pre-activations are Gaussian}
    \end{column}
    \begin{column}{0.73\textwidth}
      \[
        \vec{\lambda} = \frac{W \vec{x}}{\sqrt{d}} \in \mathbb{R}^p,
        \qquad
        \vec{\lambda}^\star = \frac{W^\star \vec{x}}{\sqrt{d}} \in \mathbb{R}^k
      \]
      \vspace{-0.7em}
      \[
        f(\vec{x};\, W, \vec{a}) = h(\vec{\lambda}, \vec{a}),
        \qquad
        h(\vec{\lambda}, \vec{a}) = \frac{1}{p}\sum_{j=1}^p a_j\,\sigma(\lambda_j)
      \]
    \end{column}
  \end{columns}
  \vspace{0.2em}
  \begin{defbox}[top=3pt, bottom=2pt]{Sufficient low-dimensional statistics}
    \fiteq{
      Q_{jl} = \frac{\langle \vec{w}_j, \vec{w}_l\rangle}{d}, \qquad
      M_{jr} = \frac{\langle \vec{w}_j, \vec{w}^\star_r\rangle}{d}, \qquad
      P_{rs} = \frac{\langle \vec{w}^\star_r, \vec{w}^\star_s\rangle}{d}}
    \fiteq{
      (\vec{\lambda}, \vec{\lambda}^\star) \sim \mathcal{N}(0, \Omega),
      \qquad
      \Omega = \begin{pmatrix} Q & M \\ M^\top & P \end{pmatrix}
        \in \mathbb{R}^{(p+k) \times (p+k)}}
  \end{defbox}
  \vspace{0.2em}
  \begin{columns}[c]
    \begin{column}{0.27\textwidth}
      \raggedleft
      \role{\large Population risk is $d$-independent}
    \end{column}
    \begin{column}{0.73\textwidth}
      \[
        \mathcal{R}(\Omega, \vec{a})
        = \mathbb{E}_{(\vec{\lambda}, \vec{\lambda}^\star) \sim \mathcal{N}(0, \Omega)}
          \!\left[\tfrac{1}{2}
            \bigl(h(\vec{\lambda}, \vec{a}) - h^\star(\vec{\lambda}^\star)\bigr)^2\right]
      \]
    \end{column}
  \end{columns}
  \note{\notetime{9:39}\begin{notelist}
    \item the problem: in $d$ dimensions the objects we track blow up as
      $d \to \infty$
    \item Gaussian data $\Rightarrow$ the pre-activations are Gaussian too
    \item so rewrite student \emph{and} teacher as functions of $\vec\lambda$ alone
    \item \textbf{``low dimensional'' here means \emph{independent of $d$}}
    \item \textbf{so we can skip the first layer entirely: sample the
      pre-activations, never the input}
    \item $Q$ student--student, $M$ student--teacher, $P$ teacher--teacher and fixed
    \item \textbf{$P$ is often just the identity, and the theory still works fine}
    \item the risk follows: change of variable, input $\to$ pre-activation
    \item \textbf{$\to$ next: and the \emph{update rule} closes on them too}
  \end{notelist}}
\end{frame}

% === 9. The reduced dynamic: update rule ==================================
% Laid out as a four-row grid with a fixed arrow gutter at 0.40/0.06/0.54: the
% left column is everything in weight space, the right column the same object
% in statistics space.  The old prose correspondence block ("update rule for
% $W$ $\leftrightarrow$ update rule for $(Q,M,\vec a)$") is gone -- the grid
% says it -- and its two remaining lines became rows 3 and 4.
% The $\lVert\vec x^\nu\rVert^2/d$ term is tinted to match the asterisk in the
% headline: it is the one thing that is not exactly self-closed, and it
% concentrates to 1 in high dimension.  Spoken, not written.
\begin{frame}{The reduced dynamic: update rule}
  \begin{center}\large
    The update rule is self-closed too!{\color{slideorange}*}
  \end{center}
  \vspace{0.2em}
  \begin{columns}[c]
    \begin{column}{0.40\textwidth}
      \centering\role{\large In the weights}
    \end{column}
    \begin{column}{0.06\textwidth}\end{column}
    \begin{column}{0.54\textwidth}
      \centering\role{\large In the sufficient statistics}
    \end{column}
  \end{columns}
  \vspace{0.2em}
  \begin{columns}[c]
    \begin{column}{0.40\textwidth}
      \fiteq{
        \begin{aligned}
          \vec{w}_j^{\nu+1} &= \vec{w}_j^{\nu}
            + \frac{\gamma}{p\sqrt{d}}
              \left( y^\nu(\vec{\lambda}^{\star,\nu}) - h(\vec{\lambda}^{\nu}) \right)
              a_j \sigma'(\lambda^\nu_j) \vec{x}^\nu, \\
          a_j^{\nu+1} &= a_j^{\nu}
            + \frac{\gamma}{pd}
              \left( y^\nu(\vec{\lambda}^{\star,\nu}) - h(\vec{\lambda}^{\nu}) \right)
              \sigma(\lambda^\nu_j)
        \end{aligned}}
    \end{column}
    \begin{column}{0.06\textwidth}
      \centering$\Longrightarrow$
    \end{column}
    \begin{column}{0.54\textwidth}
      \fiteq{
        \begin{aligned}
          Q^{\nu+1}_{jl} =\;& Q^{\nu}_{jl}
            + \frac{\gamma}{pd}
              \left( y^\nu(\vec{\lambda}^{\star,\nu}) - h(\vec{\lambda}^{\nu}) \right)
              \left( a_l \sigma'(\lambda^\nu_l) \lambda^\nu_j
                   + a_j \sigma'(\lambda^\nu_j) \lambda^\nu_l \right) + \\
            &+ \frac{\gamma}{pd}\frac{\gamma}{p}
              \left( y^\nu(\vec{\lambda}^{\star,\nu}) - h(\vec{\lambda}^{\nu}) \right)^2
              a_j a_l \sigma'(\lambda^\nu_j) \sigma'(\lambda^\nu_l)
              {\color{slideorange}\frac{\lVert\vec{x}^\nu\rVert^2}{d}}, \\
          M^{\nu+1}_{jr} =\;& M^{\nu}_{jr}
            + \frac{\gamma}{pd}
              \left( y^\nu(\vec{\lambda}^{\star,\nu}) - h(\vec{\lambda}^{\nu}) \right)
              a_j \sigma'(\lambda^\nu_j) \lambda^{\star,\nu}_r, \\
          a_j^{\nu+1} =\;& a_j^{\nu}
            + \frac{\gamma}{pd}
              \left( y^\nu(\vec{\lambda}^{\star,\nu}) - h(\vec{\lambda}^{\nu}) \right)
              \sigma(\lambda^\nu_j).
        \end{aligned}}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \begin{columns}[c]
    \begin{column}{0.40\textwidth}
      \centering
      sample $\vec{x}^\nu \sim \mathcal{N}(0, I_d)$
    \end{column}
    \begin{column}{0.06\textwidth}
      \centering$\Longrightarrow$
    \end{column}
    \begin{column}{0.54\textwidth}
      \centering
      sample $(\vec{\lambda}^\nu, \vec{\lambda}^{\star,\nu}) \sim \mathcal{N}(0, \Omega)$
    \end{column}
  \end{columns}
  \vspace{0.3em}
  \begin{columns}[c]
    \begin{column}{0.40\textwidth}
      \centering
      population risk $\mathcal{R}(W)$
    \end{column}
    \begin{column}{0.06\textwidth}
      \centering$\Longrightarrow$
    \end{column}
    \begin{column}{0.54\textwidth}
      \centering
      population risk $\mathcal{R}(Q, M, \vec{a})$
    \end{column}
  \end{columns}
  \vspace{2em}
  \punchline{There is no need to work with $W$ --- let's look at $Q, M, P$}
  \note{\notetime{11:19}\begin{notelist}
    \item rearrange the weight updates $\Rightarrow$ update rules for
      $Q$, $M$, $\vec{a}$
    \item and they \emph{close}
    \item \textbf{the orange${}^{*}$: the $\lVert\vec{x}\rVert^2/d$ term, the one
      thing not exactly self-closed --- it concentrates to $1$ as $d \to \infty$}
    \item so forget the weights: sample from $\Omega$, update the statistics
    \item every object in $d$ dimensions has a low-dimensional counterpart
    \item \textbf{now the magic: rearrange so an infinitesimal ratio sits on the
      left, and the right-hand side averages hard $\to$ its expected value}
    \item \textbf{this \emph{is} a thermodynamic limit --- the closest to physics
      this talk gets; after here it is computer science}
    \item \textbf{$\to$ next: what that limit actually produces}
  \end{notelist}}
\end{frame}

% === 10. The magic of high-dimensional data ===============================
% Split out of the old single frame, which crushed the plateau figures to a
% quarter width and the ODEs to a six-line \fiteq.  This half carries the
% equations only, written in the compact form frame 16 already uses
% (dQ/dt = Phi + gamma/p Psi) with the four expectations demoted to a

\begin{frame}{The magic of high-dimensional data}
  \begin{center}
    \large The stochastic process concentrates into a set of ODEs!
    \quad\cit{Saad \& Solla, 1995}
  \end{center}
  \vspace{0.1em}
  \begin{columns}[c]
    \begin{column}{0.56\textwidth}
      \[
        \begin{aligned}
          \frac{\dd Q_{jl}}{\dd t} &= \Phi^{Q}_{jl}(\Omega, \vec{a})
            + \frac{\gamma}{p}\,\Psi^{Q}_{jl}(\Omega, \vec{a}), \\[0.2em]
          \frac{\dd M_{jr}}{\dd t} &= \Phi^{M}_{jr}(\Omega, \vec{a}), \\[0.2em]
          \frac{\dd a_{j}}{\dd t} &= \Phi^{a}_{j}(\Omega, \vec{a}).
        \end{aligned}
      \]
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \role{\large The time-step}
      \[ \delta t = \frac{\gamma}{pd} \]
      \vspace{-0.6em}
      \par
      \aside{infinitesimal in high dimension:
        $\nu \longmapsto t = \nu\,\dfrac{\gamma}{pd}$}
    \end{column}
  \end{columns}
  \vspace{0.15em}
  \begin{defbox}[top=3pt, bottom=2pt]{Drift $\Phi$ and noise $\Psi$
      \titleaside{expectation over
        $(\vec{\lambda}, \vec{\lambda}^\star) \sim \mathcal{N}(0, \Omega)$}}
    \fiteq{
      \begin{aligned}
        \Phi^{Q}_{jl} &= \mathbb{E}\left[
          \left( h^\star(\vec{\lambda}^{\star}) - h(\vec{\lambda}) \right)
          \left( a_l \sigma'(\lambda_l) \lambda_j
               + a_j \sigma'(\lambda_j) \lambda_l \right)\right],
        &\qquad
        \Phi^{M}_{jr} &= \mathbb{E}\left[
          \left( h^\star(\vec{\lambda}^{\star}) - h(\vec{\lambda}) \right)
          a_j \sigma'(\lambda_j) \lambda^{\star}_r \right], \\[0.3em]
        \Psi^{Q}_{jl} &= \mathbb{E}\left[
          \left(\left( h^\star(\vec{\lambda}^{\star}) - h(\vec{\lambda}) \right)^2
            + \Delta\right) a_j a_l \sigma'(\lambda_j) \sigma'(\lambda_l) \right],
        &\qquad
        \Phi^{a}_{j} &= \mathbb{E}\left[
          \left( h^\star(\vec{\lambda}^{\star}) - h(\vec{\lambda}) \right)
          \sigma(\lambda_j) \right].
      \end{aligned}}
  \end{defbox}
  \note{\notetime{12:30}\begin{notelist}
    \item the discrete stochastic process $\to$ deterministic ODEs, same dynamics
    \item the terms are expectations of functions of the pre-activations
    \item \textbf{they depend on the architecture --- $\sigma$, $h^\star$, the
      realization; they are not universal}
    \item $\delta t = \gamma/(pd)$ carries discrete time into continuous time
    \item \textbf{infinitesimal precisely \emph{because} $d \to \infty$}
    \item $\Phi$, the drift: carries the dynamics towards learning
    \item $\Psi$, the noise: \textbf{a perturbation that survives the limit and
      stops you reaching the bottom of the loss}
    \item \textbf{$\to$ next: and it really is exact --- here are the plots}
  \end{notelist}}
\end{frame}

% === 14. Mean field limit =================================================
% Standard mean field: background, not this thesis -- hence no chapter tag; the
% tag starts on the next frame, which is where the contribution starts.  The
% frame is built around the one idea that makes the PDE possible: the network
% sees the weights only through their average, so every neuron is transported
% by the same field and no pairwise interaction is needed.
\begin{frame}{Mean field limit}
  \begin{center}
    \aside{$p \to \infty$ at fixed $d$, $\gamma = \mathcal{O}(1)$}
  \end{center}
  \vspace{0.1em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \centering
      \role{The network sees only the average}
      \fiteq{\hat{\mu}_p = \frac{1}{p}\sum_{i=1}^p \delta_{\vec{w}_i},
        \qquad
        \hat{f}_\mu(\vec{x}) = \int \sigma(\vec{w}^\top\vec{x})\,\dd\mu(\vec{w})}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \role{so every neuron feels the same field}
      \fiteq{
        \varphi(\vec{w}, \mu)
          = \mathbb{E}_{\vec{x}}
            \left[\sigma'(\vec{w}^\top\vec{x})\,\vec{x}
              \left(f^\star(\vec{x}) - \hat{f}_\mu(\vec{x})\right)\right]}
    \end{column}
  \end{columns}
  \vspace{0.2em}
  \begin{center}
    \aside{no pairwise interaction to track: one common velocity field
      transports the whole density}
  \end{center}
  \vspace{0.2em}
  \fiteq{\partial_t \mu_t
    = \nabla_{\vec{w}} \cdot \bigl(\mu_t\, \varphi(\cdot, \mu_t)\bigr)}
  \begin{center}
    \aside{mass conservation, nothing more}\quad
    \cit{Mei, Montanari \& Nguyen 2018; Chizat \& Bach 2018}
  \end{center}
  \vspace{0.3em}
  \punchline{\ldots{} but $\mu$ still lives on $\mathbb{R}^{d}$.}
  \note{\notetime{16:56}\begin{notelist}
    \item write the weights as a sum of deltas, the network as an integral over it
    \item $p \to \infty$: the notation survives, the target becomes an integral
    \item \textbf{with many neurons the \emph{pairwise} interactions stop
      contributing --- so there is no $Q$ left to track}
    \item every particle rides the same field; the field depends on $\mu$, but it is
      one single field
    \item mass conservation on it $\Rightarrow$ the transport PDE, nothing more
    \item \emph{punchline} --- but $\mu$ still lives on $\mathbb{R}^d$
    \item \textbf{$\to$ next: we traded $p$ for $d$, so no dimension independence
      yet. That is exactly what the next slide fixes}
  \end{notelist}}
\end{frame}

% === 18. Can we use SS for sample complexity? =============================

\begin{frame}{Can we use SS for sample complexity?}

    {\large one single statistic carries the whole dynamics}
  \end{center}
  \vspace{-0.2em}
  \fiteq{
    \frac{\dd m}{\dd t} = (1 - m^2)\,C'(m),
    \qquad
    \mathcal{R}(m) = \mathrm{const} - C(m),
    \qquad

  \vspace{0.2em}

    \begin{column}{0.48\textwidth}

        \centering
        $\displaystyle m = \frac{\langle \vec{w}, \vec{w}^\star\rangle}{d}
          \geq \eta$
        \qquad

      \end{defbox}
    \end{column}

      \par\vspace{0.35em}

      $\displaystyle m(0) \sim \frac{1}{\sqrt{d}}$
    \end{column}
  \end{columns}

    Saad \& Solla equations.
  \end{center}

    $\displaystyle C e^{CT}\sqrt{\frac{\gamma}{pd}}$\,,\quad

  \end{center}
  \note{\notetime{24:32}\begin{notelist}

    \item \textbf{so the hard part is the \emph{search}} --- weak recovery time

% === 19. The information exponent =========================================
% Reordered into an argument: define it, show what it costs, then say what it
% does not cover -- the multi-index case (leap) and the algorithmic floor

\begin{frame}{The information exponent
    {\normalfont\cit{Ben Arous, Gheissari \& Jagannath, 2021}}}
  \begin{defbox}[top=2pt, bottom=0pt]{Information Exponent}
    \fiteq{
      \ell = \min\Bigl\{ j \geq 1 : a_j = \frac{1}{j!}\,
        \mathbb{E}_{z\sim\mathcal{N}(0,1)}[h^\star(z)\,\mathrm{He}_j(z)]
        \neq 0 \Bigr\}}
  \end{defbox}
  \vspace{0.1em}
  \begin{center}
    \aside{$h^\star(z) = z^2$ gives $\ell = 2$;\qquad
      $h^\star(z) = \mathrm{He}_j(z)$ gives $\ell = j$}
  \end{center}
  \pause
  \punchline{The guiding quantity for everything that follows.}
  \vspace{0.05em}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \begin{thmbox}[top=3pt, bottom=2pt]{Theorem}
        \centering
        The sample complexity is
        \vspace{-0.8em}
        \[
          \small
          n = \begin{cases}
            \mathcal{O}(d) & \ell = 1, \\
            \mathcal{O}(d\log d) & \ell = 2, \\
            \mathcal{O}(d^{\ell-1}) & \ell \geq 3.
          \end{cases}
        \]
      \end{thmbox}
    \end{column}
    \pause
    \begin{column}{0.54\textwidth}
      % nudge the text down so its first line starts level with the top edge of
      % the theorem box, not with the box's own first baseline
      \vspace{1.05em}
      \role{\large It is not the whole story}
      \par\vspace{0.25em}
      \role{Beyond one direction:} the \emph{leap} exponent carries it to
      multi-index targets, $k > 1$ \cit{Abbe et al., 2023}
      \par\vspace{0.45em}
      \role{Not the floor either:} a partial-trace algorithm reaches
      $\mathcal{O}(d^{\ell/2})$ \cit{Anandkumar et al., 2017}
    \end{column}
  \end{columns}

    \item information exponent $=$ leading order of the Hermite expansion of

    \item \textbf{same object as the leading order of the ODE two slides ago, and

    \item theorem: $d$, then $d\log d$, then $d^{\ell-1}$
    \item \emph{pause}
    \item not the whole story: the leap exponent (Abbe) covers $k > 1$

% === 21. How this work? ===================================================
% First half of the old frame 24: the question (SGD only ever sees y linearly)
% and the exponent that answers it.  The mechanism moved to the next frame --
% the two were being read in parallel columns when they are consecutive steps
% of one argument.
\begin{frame}{How does this work?}
  \chaptertag{Chapter 5}
  \begin{center}
    {\large One-pass SGD is linear in the label}\quad
    $\displaystyle \nabla_{\vec{w}}\mathcal{R}
      = -\,\mathbb{E}\left[y\,\sigma'(\lambda)\,\frac{\vec{x}}{\sqrt{d}}\right]$
  \end{center}
  \vspace{0.2em}
  \begin{center}
    \large What if we first transform the label, and query
    $\mathbb{E}\left[g(y)\,\sigma'(\lambda)\,\frac{\vec{x}}{\sqrt{d}}\right]$?
  \end{center}
  \vspace{0.3em}
  \pause
  \begin{defbox}[top=2pt, bottom=0pt]{Generative Exponent
      \titlecit{Damian, Pillaud-Vivien, Lee \& Bruna, 2024}}
    \fiteq{
      \ell^\star = \min\Bigl\{ j \geq 1 :
        \mathbb{E}_{z\sim\mathcal{N}(0,1)}
          \bigl[g(h^\star(z))\,\mathrm{He}_j(z)\bigr]
        \neq 0 \textit{ for some } g : \mathbb{R} \to \mathbb{R} \Bigr\}}
  \end{defbox}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \centering
      $\mathrm{He}_{2k+1}$\par
      \aside{$\ell = 2k+1$, $\ell^\star = 1$}
    \end{column}
    \begin{column}{0.33\textwidth}
      \centering
      $\mathrm{He}_{2k}$\par
      \aside{$\ell = 2k$, $\ell^\star = 2$}
    \end{column}
    \begin{column}{0.33\textwidth}
      \centering
      $\mathrm{sign}(z_1 z_2 z_3)$\par
      \aside{$\ell = \ell^\star = 3$}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \punchline{$\ell^\star \le \ell$ always, and often far smaller.}
  \note{\notetime{29:23}\begin{notelist}
    \item why does repeating help? \textbf{because one-pass SGD only ever queries
      the label \emph{linearly}}
    \item so: what if we transform the label first?
    \item \emph{pause}
    \item generative exponent: the same definition, minimised over all
      transformations $g$
    \item odd Hermites give $\ell^\star = 1$, even ones give $2$ ---
      \textbf{so any Hermite polynomial is at most two}
    \item $\mathrm{sign}(z_1z_2z_3)$: no gain at all, $\ell = \ell^\star = 3$
    \item \emph{punchline} --- \textbf{$\ell^\star \leq \ell$: say ``lower
      \emph{or equal}'', the third example is the equality case}
    \item \textbf{$\to$ next: and this is where the displaced gradient comes back
      --- it produces $g$ on its own}
  \end{notelist}}
\end{frame}

% === 22. The displaced gradient synthesizes g =============================
% Second half of the old frame 24: why the displaced gradient reaches the
% generative exponent without being told which g to use.
\begin{frame}{Why does the displaced gradient reach it?}
  \chaptertag{Chapter 5}
  \begin{center}
    \large The double gradient induces a transformation
  \end{center}
  \vspace{-0.2em}
  \fiteq{
    \nabla_{\vec{w}}\ell\bigl(f(\vec{x};\tilde{\vec{w}}), y\bigr)
      = -\,y\,\sigma'(\tilde{\lambda})\,\frac{\vec{x}}{\sqrt{d}},
    \qquad
    \tilde{\lambda} = \frac{\langle\tilde{\vec{w}}, \vec{x}\rangle}{\sqrt{d}}
      = \lambda + \rho\,y\,\sigma'(\lambda)}
  \begin{center}
    \aside{expand $\sigma'$ around $\lambda$:}
  \end{center}
  \vspace{-0.4em}
  \fiteq{
    \nabla_{\vec{w}}\ell\bigl(f(\vec{x};\tilde{\vec{w}}), y\bigr)
      \approx -\,\frac{\vec{x}}{\sqrt{d}}
        \left(y\,\sigma'(\lambda) + \rho\,y^2\,\sigma'(\lambda)\sigma''(\lambda)
          + \frac{\rho^2}{2}\,y^3\,\sigma'(\lambda)^2\sigma'''(\lambda)
          + \cdots\right)}
  \vspace{-0.1em}
  \pause
  \punchline{Powers of $y$ appear on their own: the network builds $g$ for free.}
  \vspace{0.4em}
  \begin{columns}[c]
    \begin{column}{0.62\textwidth}
      \begin{thmbox}[top=3pt, bottom=2pt]{Theorem (Repetita Iuvant)}
        \centering
        Sample complexity of EGD
        \vspace{-0.8em}
        \[
          \small
          n = \begin{cases}
            \mathcal{O}(d) & \ell^\star = 1, \\
            \mathcal{O}(d\log d) & \ell^\star = 2.
          \end{cases}
        \]
      \end{thmbox}
    \end{column}
    \begin{column}{0.38\textwidth}
      \aside{no results for $\ell^\star > 2$: we worked on polynomials}
    \end{column}
  \end{columns}
  \note{\notetime{30:26}\begin{notelist}
    \item displace the gradient: still a linear correlation, but the pre-activation
      is evaluated slightly off-centre
    \item Taylor expand $\sigma'$ around $\lambda$
    \item powers of $y$ appear on their own --- \textbf{the gradient is no longer
      linear in $y$, it has become a polynomial transformation}
    \item \emph{pause}
    \item \textbf{so the network synthesizes $g$ for free, without being told which
      one --- provided $\sigma$ is rich enough in coefficients}
    \item theorem (\emph{Repetita Iuvant}): $\ell^\star = 1 \to d$,
      $\ell^\star = 2 \to d\log d$
    \item \textbf{only up to $\ell^\star = 2$, because we work with polynomials ---
      and polynomials have \emph{generative} exponent at most two}
    \item \textbf{$\to$ next: enough two-layer networks. It is 2026}
  \end{notelist}}
\end{frame}

% === 23. IE & Attention model =============================================
% The SIE is the frame's subject, so it takes the full width instead of being
% squeezed into a 0.56 column where \fiteq shrank it to scriptsize; examples
% and theorem sit side by side underneath it.
% The theorem box used to be titled "Repetita Iuvant" and to quote the sample
% complexity of EGD -- both carried over from the Chapter 5 frame.  This is
% Theorem 2506:thm:sie, and it is about plain SGD.
\begin{frame}{IE \& Attention model}
  \chaptertag{Chapter 6}
  \adjustbox{max width=\linewidth}{%
    {\large Our proxy for attention:}\;
    $\displaystyle
      f(X) = \vec{a}_L^\top \SoftMax\!\left(\left(X + \frac{P}{\sqrt{d}}\right)
        \vec{w}\vec{w}^\top \left(X + \frac{P}{\sqrt{d}}\right)^\top\right)\vec{a}_R$
    \; $X \in \mathbb{R}^{L \times d}$}
  \begin{center}
    % slashed rather than \displaystyle: a stacked fraction here makes this
    % single line as tall as three, at the figures' expense
    {\large Sufficient statistics:}\quad
    {\large $m = \vec{w}_\star^\top\vec{w}/d$
      \quad and \quad $\vec{e} = P\vec{w}/\sqrt{d}$}
  \end{center}
  \pause
  \begin{defbox}[top=2pt, bottom=0pt]{Sequence Information Exponent (SIE)}
    \fiteq[0.82\linewidth]{
      \mathit{SIE}(\vec{f}^{\,\mathrm{SSI}}_{\vec{w}_\star}) \coloneqq
        \min\left\{\sum_{l=1}^L k_l > 0 : \vec{k} \in \mathbb{N}^L,\;
          \mathbb{E}_{\vec{z}\sim\mathcal{N}(0,I_L)}\left[
            \left(\prod_{l=1}^L \mathrm{He}_{k_l}(z_l)\right) h^\star(\vec{z})
          \right] \neq 0 \right\}}
  \end{defbox}
  \vspace{0.1em}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \emph{Examples}:
      \begin{itemize}
        \footnotesize
        \setlength{\itemsep}{0em}
        \setlength{\parskip}{0em}
        \item $z_{\star,1} + \dots + z_{\star,L}$ has SIE $= 1$;
        \item $z_{\star,1}z_{\star,2}$ has SIE $= 2$;
        \item $\prod_{l=1}^L \mathrm{He}_{k_l}(z_{\star,l})$
          has SIE $= \sum_{l=1}^L k_l$.
      \end{itemize}
    \end{column}
    \pause
    \begin{column}{0.45\textwidth}
      \vspace{-1.1em}
      \begin{thmbox}[top=3pt, bottom=2pt]{Theorem}
        \centering
        Sample complexity of SGD
        \vspace{-1.0em}
        \[
          \footnotesize
          \begin{cases}
            \mathcal{O}_L(d) & \textit{if } \mathit{SIE} = 1 \\
            \mathcal{O}_L(d\log d) & \textit{if } \mathit{SIE} = 2 \\
            \mathcal{O}_L(d^{\mathit{SIE}-1}) & \textit{if } \mathit{SIE} \geq 3
          \end{cases}
        \]
      \end{thmbox}
    \end{column}
  \end{columns}
  \vspace{0.05em}
  \punchline{The same table as before, with SIE in place of $\ell$.}
  \note{\notetime{32:10}\begin{notelist}
    \item \textbf{everyone uses transformers --- at some point in the PhD we had to
      do something about it}
    \item so: model attention only. \textbf{Arguably the core part of a transformer}
    \item simplifications: no value matrix, tied key and query, one direction, no
      heads
    \item \textbf{but the positional encoding is in}
    \item the input is now $L$ tokens of dimension $d$; $d \to \infty$ as before
    \item statistics: $m$, the teacher overlap, and $\vec{e}$, the positional one
    \item \emph{pause}
    \item SIE: break the target into multi-dimensional Hermites, take the lowest
      total order
    \item examples: a sum of tokens gives $1$; a product of two gives $2$; a product
      of Hermites gives the sum of the degrees
    \item \emph{pause} --- \textbf{the same table as Ben Arous, with SIE in place of
      $\ell$}
  \end{notelist}}
\end{frame}

          \titlecit{Abbe, Boix-Adser\`a \& Misiakiewicz, 2023}}
        \centering
        \aside{effective exponent, given the already-recovered set $S$:}
        \vspace{-0.5em}
        \fiteq{\ell^{S}_{\vec{v}} =
          \min\Bigl\{\textstyle\sum_{r:\,\vec{w}^\star_r \notin S}\alpha_r
            \;:\; a_\alpha \neq 0,\ \alpha_{\vec{v}} > 0\Bigr\}}
        \vspace{-0.5em}
        \fiteq{\mathrm{Leap} = \min_{\pi \in \mathfrak{S}_k}\
          \max_{r \in [k]}\ \ell^{\,S_{r-1}}_{\vec{w}^\star_{\pi(r)}}}
        \centering
        \aside{best order to learn in, worst step along it}
      \end{defbox}
      \vspace{0.1em}
      \punchline{$n = \mathcal{O}(d^{\mathrm{Leap}-1})$, not
        $\mathcal{O}(d^{\ell-1})$.}
      \vspace{0.25em}
      \begin{center}
        \footnotesize
        $\lambda^\star_1 + \lambda^\star_1\lambda^\star_2
          + \lambda^\star_1\lambda^\star_2\lambda^\star_3$:\;
        each recovered overlap turns the rung above it linear: three times
        $\mathcal{O}(d)$. The bare
        $\lambda^\star_1\lambda^\star_2\lambda^\star_3$ has
        $\mathrm{Leap} = 3$: $\mathcal{O}(d^2)$.
        \par\vspace{0.35em}
        \aside{$\min_{\vec{v}}\ell_{\vec{v}} = 1$ is not enough:\;
          $\lambda^\star_1 + \lambda^\star_2\lambda^\star_3\lambda^\star_4$
          and
          $\lambda^\star_1 + \lambda^\star_1\lambda^\star_2\lambda^\star_3\lambda^\star_4$
          both have $\mathrm{Leap} = 3$.}
      \end{center}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth, height=0.70\textheight, keepaspectratio]%
        {chapters/figs/staircase/staircase}
      \par\vspace{-0.2em}
      \aside{dots: online SGD;\; lines: the Saad \& Solla equations}
    \end{column}
  \end{columns}
  \note{\notebackup{Chapter 5}\begin{notelist}
    \item frame 19 said ``the leap exponent carries it to $k > 1$'' and moved
      on --- this is what that actually means
    \item at initialization every overlap is $1/\sqrt{d}$, so every term of the
      expansion vanishes: the landscape is flat in \emph{all} directions
    \item once one direction is recovered its overlap is $\mathcal{O}(1)$ and
      acts as a \textbf{constant multiplier}, lowering the effective degree of
      every monomial that contains it
    \item so the dynamics goes saddle to saddle, and the cost is set by the
      \emph{worst} step, minimized over the order you climb in
    \item \textbf{the example: $\lambda^\star_1\lambda^\star_2$ reads as
      $m_1\lambda^\star_2$ once $m_1$ is finite --- a linear signal in the
      second direction}
    \item plot, solid: $m_1$ then $m_2$ then $m_3$, three visible steps;
      dashed is the bare monomial and it \textbf{never leaves zero}
    \item \textbf{the two counterexamples are the point of the definition:
      having one linear direction does not make the target easy if the rest is
      disconnected from it}
    \item \textbf{why you jump here: someone asks what the leap exponent is, or
      why the information exponent is not enough for $k > 1$}
  \end{notelist}}
\end{frame}

% === B10. Why polynomials? ================================================
% Frame 22 ends on a dangling aside ("no results for l* > 2: we worked on
% polynomials"), which reads like a missing case.  It is not.
%
% The frame is built on the one observation that makes both halves of the
% answer obvious at once: all three exponents are the SAME minimum, taken over
% a wider and wider class of output transformations.  So the chain
% l* <= l*_p <= l is not a fact to memorise -- it is monotonicity of a min over
% nested classes -- and "why polynomials" becomes the single question of which
% rung the displaced gradient can actually reach.
%
% The bottom half then answers the two questions the ladder provokes, which are
% the two the examiner will actually ask: the expansion is a SERIES, so how can
% it stand in for a polynomial -- and why is the theorem's guarantee only for
% almost every rho_0.  Both come out of the same object, the correlation
% F(rho_0), so they get one column each and share the equation between them.
%
% No example on the frame.  A target with l* < l*_p strictly would need a
% moment-indeterminate label law -- infinitely oscillating conditional
% expectations -- and no such link is presentable; conversely for every
% monotone, even-unimodal, bounded or at-most-quadratic-growth link the two
% coincide.  Saying "the restriction costs nothing here" is the stronger
% answer, and it lives in the notes rather than costing a line on the slide.
\begin{frame}{Why polynomials? What EgD can reach}
  \chaptertag{Chapter 5}
  \vspace{-0.3em}
  \begin{center}
    \adjustbox{max width=\linewidth}{%
      \begin{tabular}{@{}r@{\;}c@{\;}l@{\qquad}l@{}}
        \role{$\ell$} & $=$ &
          $\min\{k : \mathbb{E}[\,y\,\mathrm{He}_k(z)] \neq 0\}$ &
          \aside{no transformation: one-pass SGD} \\[0.3em]
        \role{$\ell^\star_p$} & $=$ &
          $\min\{k : \mathbb{E}[\,p(y)\,\mathrm{He}_k(z)] \neq 0\}$ &
          \aside{$p$ polynomial: \textbf{the rung EgD reaches}} \\[0.3em]
        \role{$\ell^\star$} & $=$ &
          $\min\{k : \mathbb{E}[\,g(y)\,\mathrm{He}_k(z)] \neq 0\}$ &
          \aside{any $g$ at all: the SQ optimum \cit{Damian et al.\ '24}}
      \end{tabular}}
  \end{center}
  \vspace{0.1em}
  \punchline{Wider class, smaller minimum:\;
    $\ell^\star \leq \ell^\star_p \leq \ell$.}
  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \centering
      \role{We are handed a \emph{series}, not a $p$}
      \fiteq{g_{\rho_0}(y) = \sum_{j \geq 0} c_j\,\rho_0^{\,j}\,y^{\,j+1}}
      \vspace{-0.5em}
      \aside{nothing is truncated and nothing is chosen either:
        $c_j \propto \mathbb{E}\bigl[\sigma^{(j+1)}\sigma'^{\,j}\bigr]$ is
        fixed by the activation alone.}
    \end{column}
    \begin{column}{0.53\textwidth}
      \centering
      \role{but its reach is exactly $\ell^\star_p$}
      \fiteq{\mathbb{E}\bigl[g_{\rho_0}(y)\,\mathrm{He}_k(z)\bigr]
        = \sum_{m \geq 1} c_{m-1}\,\rho_0^{\,m-1}\,\alpha_m}
      \vspace{-0.5em}
      \aside{the same $\alpha_m = \mathbb{E}[y^m\mathrm{He}_k(z)]$ that define
        $\ell^\star_p$. All zero $\Rightarrow$ no better than a polynomial;
        one non-zero $\Rightarrow$ no worse.}
    \end{column}
  \end{columns}
  \vspace{0.25em}
  \begin{center}
    \aside{\textbf{And that is the ``almost every $\rho_0$''}:\; analytic in
      $\rho_0$, not identically zero $\Rightarrow$ isolated zeros.}
  \end{center}
  \vspace{0.25em}
  \begin{center}
    \begin{minipage}{0.82\textwidth}
      \begin{thmbox}[top=3pt, bottom=2pt]{Theorem}
        \centering
        If $h^\star$ is a polynomial, then $\ell^\star_p \leq 2$.
        If $\ell^\star_p = 1$ unless $h^\star$ is even.
      \end{thmbox}
    \end{minipage}
  \end{center}
  \note{\notebackup{Chapter 5}\begin{notelist}
    \item frame 22's footnote --- \textbf{it sounds like a gap in the proof and
      it is not}
    \item top: \textbf{one minimum, a wider and wider class of $g$} --- $\ell$
      plain SGD, $\ell^\star$ any $g$ (Damian), $\ell^\star_p$ ours; the chain
      is monotonicity
    \item \textbf{the objection: expanding $\sigma'$ gives a series, not a
      polynomial.} Nothing is truncated, and we never pick the $c_j$:
      $\sigma$ fixes them, non-zero for a.e.\ bias shift
    \item \textbf{but only reach matters} --- correlate against $\mathrm{He}_k$
      and out come the same $\alpha_m$ that define $\ell^\star_p$: all zero,
      the series is zero too; one non-zero, it is a non-zero \emph{analytic}
      function of $\rho_0$ --- \textbf{whose zeros are isolated, and that is
      the ``almost every $\rho_0$''}
    \item \textbf{theorem closes that rung at two}: $p(y) = y^m$ with odd $m$,
      Gaussian moments beat any $C^m$; even targets test $z^2 - 1$.
      \textbf{No missing $\ell^\star_p = 3$ case}
    \item \textbf{does the restriction ever cost anything?} No --- the two
      agree for every monotone, bounded or quadratic-growth link; a strict gap
      needs moment-indeterminate labels
    \item \textbf{why you jump here: someone asks why the theorem stops at
      $\ell^\star = 2$}
  \end{notelist}}
\end{frame}

  % No defbox around the operator: its title bar, padding and shadow cost about

  \begin{center}
    \role{One operator per layer}\quad
    \aside{the label-weighted, Hessian-like term of layerwise GD at vanishing
      initialization}
  \end{center}
  \vspace{-0.6em}
  \fiteq[0.54\linewidth]{\widehat{C}^{(\ell)} = \frac{1}{n}\sum_{\mu=1}^{n}
    y_\mu\, \vec{z}_{\ell-1}(\vec{x}_\mu)\,
    \vec{z}_{\ell-1}(\vec{x}_\mu)^{\top}}
  \vspace{-0.4em}
  % The three steps on one line, scaled down only if they overflow -- as a
  % three-row block they cost twice the height for no extra information.
  \centerline{\adjustbox{max width=\linewidth}{%
    \role{Filter}\; $\vec{g}_\ell = \widehat{V}_\ell^{\top}\vec{z}_{\ell-1}$\;
    \aside{top-$k$ eigenvectors}\quad$\longrightarrow$\quad
    \role{Lift}\; $\vec{z}_\ell = \sigma(R_\ell\vec{g}_\ell)/\sqrt{p_\ell}$\;
    \aside{random, nonlinear}\quad$\longrightarrow$\quad
    \role{Repeat}\; $K_0 \to K_1 \to \cdots \to K_L$}}
  \vspace{0.4em}
  \punchline{Keep what is \emph{simple} in the previous geometry and
    \emph{predictive} at low degree.}

  \end{center}
  \note{\notebackup{Beyond the thesis}\begin{notelist}
    \item \textbf{one operator, one eigendecomposition per layer --- that is the
      whole method}: the second-order term of layerwise GD at vanishing
      initialization
    \item filter keeps the top eigen-directions, lift re-expands them at random
      --- \textbf{the lift matters most: it changes the geometry, so the next
      layer sees signal that was high-degree in the input}
    \item so depth is an adaptive kernel construction: one kernel per layer,
      each chosen \emph{after} seeing the labels
    \item the figure: \textbf{LoFi beats early GD at small $n$; GD pulls ahead
      only once it gets enough steps}
    \item \textbf{tie-back: a feature emerges when $\rho \gg \tau(n)$ --- a BBP

      order is the leap exponent}
    \item \textbf{why you jump here: frame 25 promised LoFi and someone asks
      what it is} --- \textbf{$\to$ next backup: what it finds on real images}
  \end{notelist}}
