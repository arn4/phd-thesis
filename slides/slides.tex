% === 3. What does a physicist do about it? ==================================
\begin{frame}{What does a physicist do about it?}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \role{\large Why the classical theory does not apply}
      \begin{itemize}
        \setlength{\itemsep}{0.35em}
        \item fitting data with a huge number of parameters breaks the
          bias--variance trade-off
        \item non-convex landscape, no closed-form solution
        \item \dots\ yet plain SGD finds solutions that generalize
      \end{itemize}
      \vspace{0.9em}
      \role{\large So we build a caricature}\par\vspace{0.25em}
      \small Simplified and certainly incomplete, but \emph{concrete enough} to be
      tested, refuted and improved.
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \atomprogression
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \punchline{We want to be Thomson, so that someone can be Bohr after us.}
  \vspace{0.6em}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}\centering\small
      \role{Worst case} \aside{(PAC, VC, Rademacher)}\par
      $\longrightarrow$ pessimistic bounds
    \end{column}
    \begin{column}{0.5\textwidth}\centering\small
      \role{Typical case} \aside{(statistical physics)}\par
      $\longrightarrow$ sharp predictions
    \end{column}
  \end{columns}
  \note{Chemistry at the end of the nineteenth century could synthesize
    complex molecules without knowing why atoms bond: Thomson's model was crude
    and wrong in important ways, but concrete enough to be improved into Bohr,
    and eventually into quantum mechanics.  That is roughly where deep learning
    stands, and the role we aspire to play.
    Two advantages we have: the experiments run on a laptop rather than a
    collider, and the process itself -- training -- is mathematically well
    defined, so we only need to explain \emph{why} it works and \emph{how} the
    dynamics unfold.
    Last line: this is why we use statistical physics rather than classical
    learning theory, and the thread of the thesis is the typical dynamics of
    SGD in high dimension.}
\end{frame}

% === 4. Our setting =========================================================
\begin{frame}{Our setting}
  \begin{center}
    {\Large\bfseries Supervised Learning with scalar labels}
  \end{center}
  \vspace{0.3em}
  \begin{center}
    \small
    $\mathcal{D} = \{(\vec{x}^\nu,\, y^\nu)\}_{\nu \in [n]}$ i.i.d.,\quad
    $\vec{x} \in \mathbb{R}^d$ ($d \gg 1$),\quad $y \in \mathbb{R}$
    \quad$\longrightarrow$\quad
    find $f$ with $f(\vec{x}) \approx y$ on \emph{unseen} data
  \end{center}
  % \vspace{0.1em}
  \begin{center}
    \aside{The most controlled paradigm there is --- and a scalar label carries
      more signal than a class label.}
  \end{center}
  % \vspace{0.25em}
  \begin{center}
    \large We need to make some assumptions for our theory on 
  \end{center}
  \vspace{0.4em}
  \begin{columns}[t]
    \begin{column}{0.33\textwidth}\centering
      \icondata\par\vspace{0.25em}
      \role{\large The data}\par\vspace{0.2em}
      % \small Where do the inputs come from, and how is the label built?
    \end{column}
    \begin{column}{0.33\textwidth}\centering
      \iconmodel\par\vspace{0.25em}
      \role{\large The model}\par\vspace{0.2em}
      % \small Which family of functions do we fit, and how do we score it?
    \end{column}
    \begin{column}{0.33\textwidth}\centering
      \iconalgo\par\vspace{0.25em}
      \role{\large The algorithm}\par\vspace{0.2em}
      % \small How are the parameters updated, and with how much data?
    \end{column}
  \end{columns}
  \vspace{1.2em}
  \punchline{Keep only what the phenomenon needs, strip everything else.}
  \note{This is not of course our idea. It has been build on ton of literature.
    We are based on 3 pillars.
    Supervised regression with a scalar label is the most controlled paradigm
    there is: we need not lean on the complex internal structure of the data,
    and that control is exactly what a physicist needs to build and test a
    theory.
    Any supervised learning problem is three choices: a data distribution, a
    model class, and a learning algorithm.  Each one is a whole universe, and
    the single criterion for picking inside it is: keep the features that are
    essential for the phenomenon we want to study, and strip everything else.
    The next three slides answer the three questions in order.}
\end{frame}

% === 5. The multi-index models =============================================
\begin{frame}{The multi-index models}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \role{Gaussian data}\qquad $\vec{x} \sim \mathcal{N}(0, I_d)$
      \vspace{1.4em}

      \role{A proxy for feature learning}
      \[
        h^\star : \mathbb{R}^k \to \mathbb{R}
        \qquad
        f^\star(\vec{x}) = h^\star\!\left(\frac{W^\star \vec{x}}{\sqrt{d}}\right)
      \]
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      $W^\star = \{\vec{w}^\star_r\}_{r \in [k]} \in \mathbb{R}^{k \times d}$
      \par\vspace{0.5em}
      \thesisnet[0.95\linewidth]{chapters/figs/multi-index.tex}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{center}
    \role{(Possible) Label Noise}\qquad
    $y = f^\star(\vec{x}) + \sqrt{\Delta}\,\xi, \qquad \xi \sim \mathcal{N}(0,1)$
  \end{center}
  \note{$d$ is going to be be big, while $k$ is small and fixed: discovering the
    subspace $W^\star$ is actually a proxy of feature learning.}
\end{frame}

% === 6. The learned model: 2-layer network =================================
\begin{frame}{The learned model: 2-layer network}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \[
        f(\vec{x};\, W, \vec{a})
        = \frac{1}{p} \sum_{j=1}^p a_j\,
          \sigma\!\left(\frac{\langle \vec{w}_j, \vec{x}\rangle}{\sqrt{d}}\right)
      \]
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \thesisnet[0.95\linewidth]{chapters/figs/two-layer-net.tex}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{columns}[t]
    \begin{column}{0.42\textwidth}
      \centering
      \role{\large Loss function}
      \[ \ell(\hat{y}, y) = \tfrac{1}{2}(\hat{y} - y)^2 \]
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering
      \role{\large Population risk}
      \[
        \mathcal{R}(W, \vec{a}) = \mathbb{E}_{(\vec{x},\, y)}\!\left[
          \tfrac{1}{2}\bigl(f(\vec{x};\, W, \vec{a}) - y\bigr)^2\right]
      \]
    \end{column}
  \end{columns}
  \note{$W$, $\vec{a}$ are the weights. $\sigma$ is the activation function.
    This is the natural loss for regression.
    The population risk then is the best measure of the performance, but of
    course it can't be used for training (otherwise we have an equivalent of
    data leakage).}
\end{frame}

% === 7. The algorithm: one-pass SGD ========================================
\begin{frame}{The algorithm: one-pass SGD}
  \begin{columns}[c]
    \begin{column}{0.26\textwidth}\raggedleft\role{\large Update}\end{column}
    \begin{column}{0.74\textwidth}
      \[
        \vec{w}_j^{\,\nu+1} = \vec{w}_j^{\,\nu}
          - \gamma\,\nabla_{\vec{w}_j}\,
            \ell\!\left(f(\vec{x}^\nu;\, W^\nu, \vec{a}),\, y^\nu\right),
        \qquad j \in [p],
      \]
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \begin{columns}[c]
    \begin{column}{0.26\textwidth}\raggedleft\role{\large Initialization}\end{column}
    \begin{column}{0.74\textwidth}
      \[ \vec{w}_j^0 \sim \mathcal{N}(0, I_d) \]
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \begin{columns}[c]
    \begin{column}{0.26\textwidth}\raggedleft\role{\large The key property}\end{column}
    \begin{column}{0.74\textwidth}
      \punchline{Independence of the steps}
    \end{column}
  \end{columns}
  \vfill
  \begin{center}
    \aside{We actually often consider a spherical constrained dynamics for
      simplicity:}\quad
    $\displaystyle \vec{w}^{\nu+1} = \sqrt{d}\,
      \frac{\vec{w}^\nu - \gamma\nabla_{\vec{w}}\ell(f(\vec{x}^\nu), y^\nu)}
           {\bigl\lVert \vec{w}^\nu
             - \gamma\nabla_{\vec{w}}\ell(f(\vec{x}^\nu), y^\nu)\bigr\rVert}$
  \end{center}
  \note{$W$, $\vec{a}$ are the weights. $\sigma$ is the activation function.
    It is one pass because every sample is used once then discarded forever.}
\end{frame}

% === 8. The reduced dynamic ================================================
\begin{frame}{The reduced dynamic}
  \begin{columns}[c]
    \begin{column}{0.34\textwidth}
      \role{\large Pre-activations are Gaussian}
    \end{column}
    \begin{column}{0.66\textwidth}
      \[
        \vec{\lambda} = \frac{W \vec{x}}{\sqrt{d}} \in \mathbb{R}^p,
        \qquad
        \vec{\lambda}^\star = \frac{W^\star \vec{x}}{\sqrt{d}} \in \mathbb{R}^k
      \]
      \[
        f(\vec{x};\, W, \vec{a}) = h(\vec{\lambda}, \vec{a}),
        \qquad
        h(\vec{\lambda}, \vec{a}) = \frac{1}{p}\sum_{j=1}^p a_j\,\sigma(\lambda_j)
      \]
    \end{column}
  \end{columns}
  \vspace{0.3em}
  \punchline{Sufficient low-dimensional statistics}
  \fiteq{
    Q_{jl} = \frac{\langle \vec{w}_j, \vec{w}_l\rangle}{d}, \quad
    M_{jr} = \frac{\langle \vec{w}_j, \vec{w}^\star_r\rangle}{d}, \quad
    P_{rs} = \frac{\langle \vec{w}^\star_r, \vec{w}^\star_s\rangle}{d}
    \qquad
    (\vec{\lambda}, \vec{\lambda}^\star) \sim \mathcal{N}(0, \Omega), \quad
    \Omega = \begin{pmatrix} Q & M \\ M^\top & P \end{pmatrix}
      \in \mathbb{R}^{(p+k) \times (p+k)}}
  \begin{columns}[c]
    \begin{column}{0.38\textwidth}
      \role{\large Population risk is $d$-independent}
    \end{column}
    \begin{column}{0.62\textwidth}
      \[
        \mathcal{R}(\Omega, \vec{a})
        = \mathbb{E}_{(\vec{\lambda}, \vec{\lambda}^\star) \sim \mathcal{N}(0, \Omega)}
          \!\left[\tfrac{1}{2}
            \bigl(h(\vec{\lambda}, \vec{a}) - h^\star(\vec{\lambda}^\star)\bigr)^2\right]
      \]
    \end{column}
  \end{columns}
  \note{Low dimensional aka independent of $d$.}
\end{frame}

% === 9. The reduced dynamic: update rule ===================================
\begin{frame}{The reduced dynamic: update rule}
  \begin{center}\large Update rule is self-closed too!*\end{center}
  \vspace{0.3em}
  \begin{columns}[c]
    \begin{column}{0.40\textwidth}
      \fiteq{
        \begin{aligned}
          \vec{w}_j^{\nu+1} &= \vec{w}_j^{\nu}
            + \frac{\gamma}{p\sqrt{d}}
              \left( y^\nu(\vec{\lambda}^{\star,\nu}) - h(\vec{\lambda}^{\nu}) \right)
              a_j \sigma'(\lambda^\nu_j) \vec{x}^\nu, \\
          a_j^{\nu+1} &= a_j^{\nu}
            + \frac{\gamma}{pd}
              \left( y^\nu(\vec{\lambda}^{\star,\nu}) - h(\vec{\lambda}^{\nu}) \right)
              \sigma(\lambda^\nu_j)
        \end{aligned}}
    \end{column}
    \begin{column}{0.06\textwidth}
      \centering$\Longrightarrow$
    \end{column}
    \begin{column}{0.54\textwidth}
      \fiteq{
        \begin{aligned}
          Q^{\nu+1}_{jl} =\;& Q^{\nu}_{jl}
            + \frac{\gamma}{pd}
              \left( y^\nu(\vec{\lambda}^{\star,\nu}) - h(\vec{\lambda}^{\nu}) \right)
              \left( a_l \sigma'(\lambda^\nu_l) \lambda^\nu_j
                   + a_j \sigma'(\lambda^\nu_j) \lambda^\nu_l \right) + \\
            &+ \frac{\gamma}{pd}\frac{\gamma}{p}
              \left( y^\nu(\vec{\lambda}^{\star,\nu}) - h(\vec{\lambda}^{\nu}) \right)^2
              a_j a_l \sigma'(\lambda^\nu_j) \sigma'(\lambda^\nu_l)
              \frac{\lVert\vec{x}^\nu\rVert^2}{d}, \\
          M^{\nu+1}_{jr} =\;& M^{\nu}_{jr}
            + \frac{\gamma}{pd}
              \left( y^\nu(\vec{\lambda}^{\star,\nu}) - h(\vec{\lambda}^{\nu}) \right)
              a_j \sigma'(\lambda^\nu_j) \lambda^{\star,\nu}_r, \\
          a_j^{\nu+1} =\;& a_j^{\nu}
            + \frac{\gamma}{pd}
              \left( y^\nu(\vec{\lambda}^{\star,\nu}) - h(\vec{\lambda}^{\nu}) \right)
              \sigma(\lambda^\nu_j).
        \end{aligned}}
    \end{column}
  \end{columns}
  \vspace{0.2em}
  \[
    \small
    \begin{aligned}
      \text{update rule for } W \quad &\leftrightarrow \quad
        \text{update rule for } (Q, M, \vec{a}) \\
      \text{sample } \vec{x}^\nu \sim \mathcal{N}(0, I_d) \quad &\leftrightarrow \quad
        \text{sample } (\vec{\lambda}^\nu, \vec{\lambda}^{\star,\nu}) \sim \mathcal{N}(0, \Omega) \\
      \text{population risk } \mathcal{R}(W) \quad &\leftrightarrow \quad
        \text{population risk } \mathcal{R}(Q, M, \vec{a}).
    \end{aligned}
  \]
  \punchline{There is no need to work with $W$, let's look at $Q,M,P$}
  \note{Low dimensional aka independent of $d$.}
\end{frame}

% === 10. The magic of high-dimensional data =================================
\begin{frame}{The magic of high-dimensional data}
  \begin{center}
    \large The stochastic process concentrates into a set of ODEs!
  \end{center}
  \begin{columns}[c]
    \begin{column}{0.25\textwidth}
      \includegraphics[width=\linewidth]{chapters/figs/saad-solla-plateau/plateau-noiseless}
    \end{column}
    \begin{column}{0.25\textwidth}
      \includegraphics[width=\linewidth]{chapters/figs/saad-solla-plateau/plateau-noisy}
    \end{column}
    \begin{column}{0.50\textwidth}
      \fiteq{
        \begin{aligned}
          \frac{\dd Q_{jl}}{\dd t} =\;&
            \overbrace{\mathbb{E}_{(\vec{\lambda}, \vec{\lambda}^\star) \sim \mathcal{N}(0, \Omega)}
              \left[ \left( h^\star(\vec{\lambda}^{\star}) - h(\vec{\lambda}) \right)
                \left( a_l \sigma'(\lambda_l) \lambda_j
                     + a_j \sigma'(\lambda_j) \lambda_l \right)\right]}
              ^{\textstyle \eqqcolon \Phi^{Q}_{jl}(\Omega, a)} \\
            &+ \frac{\gamma}{p}
              \underbrace{\mathbb{E}_{(\vec{\lambda}, \vec{\lambda}^\star) \sim \mathcal{N}(0, \Omega)}
                \left[ \left(\left( h^\star(\vec{\lambda}^{\star}) - h(\vec{\lambda}) \right)^2
                  + \Delta\right) a_j a_l \sigma'(\lambda_j) \sigma'(\lambda_l) \right]}
                _{\textstyle \eqqcolon \Psi^{Q}_{jl}(\Omega, a)}, \\[0.6em]
          \frac{\dd M_{jr}}{\dd t} =\;&
            \underbrace{\mathbb{E}_{(\vec{\lambda}, \vec{\lambda}^\star) \sim \mathcal{N}(0, \Omega)}
              \left[ \left( h^\star(\vec{\lambda}^{\star}) - h(\vec{\lambda}) \right)
                a_j \sigma'(\lambda_j) \lambda^{\star}_r \right]}
              _{\textstyle \eqqcolon \Phi^{M}_{jr}(\Omega, a)}, \\[0.6em]
          \frac{\dd a_j}{\dd t} =\;&
            \underbrace{\mathbb{E}_{(\vec{\lambda}, \vec{\lambda}^\star) \sim \mathcal{N}(0, \Omega)}
              \left[ \left( h^\star(\vec{\lambda}^{\star}) - h(\vec{\lambda}) \right)
                \sigma(\lambda_j) \right]}
              _{\textstyle \eqqcolon \Phi^{a}_{j}(\Omega, a)}.
        \end{aligned}}
    \end{column}
  \end{columns}
  \vfill
  \begin{columns}[c]
    \begin{column}{0.42\textwidth}
      \centering
      \role{The time-step}\quad
      $\displaystyle \delta t = \frac{\gamma}{pd}$
      \par\vspace{0.2em}
      \aside{Infinitesimal for high dimension}
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering
      \role{The actual bound}
      \fiteq{
        \max_{0 \le \nu \le Td} \mathbb{E}\left[
          \left\lVert (\Omega^\nu, a^\nu)
            - \left(\tilde{\Omega}\!\left(\frac{\nu\gamma}{pd}\right),
                    \tilde{a}\!\left(\frac{\nu\gamma}{pd}\right)\right)\right\rVert
        \right] \le \frac{C(T)}{\sqrt{d}}}
    \end{column}
  \end{columns}
  \note{Those are erf--erf plots, the classic of committee machines.
    This is an intuition: LHT gets closer and closer to a derivative, while the
    RHS s some sort of thermodynamic limit (this is the closest I can get to
    physics in this presentation).
    A bit of taxonomy of the SS equations: one term is the drift (gradient flow
    for the population risk), while the other is an high-dimensional noise term
    that survive the limit: it is distinguishing characteristics of SS dynamic!}
\end{frame}

% === 13. Mean field limit ==================================================
\begin{frame}{Mean field limit}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \centering
      The weights ``flow'' through a field
      \fiteq{\hat{f}_\mu(\vec{x}) = \int \sigma(\vec{w}^\top\vec{x})\,\dd\mu(\vec{w})}
      \fiteq{
        \varphi(\vec{w}, \mu)
          = \mathbb{E}_{\vec{x} \sim \mathcal{N}(\vec{0}, \sfrac{1}{d}I_d)}
            \left[\sigma'(\vec{w}^\top\vec{x})\,\vec{x}
              \left(f_{\Theta^\star}(\vec{x}) - \hat{f}_\mu(\vec{x})\right)\right]}
      \vspace{0.5em}
      \fiteq{\partial_t \mu_t
        = \nabla_{\vec{w}} \cdot \bigl(\mu_t \varphi(\cdot, \mu_t)\bigr)}
      \vspace{1em}

      $\mu$ is a density over $\mathbb{R}^d$
    \end{column}
    \begin{column}{0.02\textwidth}
      \rule{0.4pt}{0.68\textheight}
    \end{column}
    \begin{column}{0.50\textwidth}
      \centering
      $Q,M$ are $\mathbb{R}^{p \times (p+k)}$\ldots{} untraceable
      \fiteq{
        \begin{aligned}
          W &= M P^{-1} W^\star + W^\bot \\
          Q^\bot &= \sfrac{1}{d}\,W^\bot (W^\bot)^\top = Q - M P^{-1} M^\top
        \end{aligned}}
      We can trace $M$ and $Q^\bot$
      \vspace{0.3em}

      \begin{ansatzbox}{Ansatz}
        \centering
        \emph{We just need the diagonal of}
        \fiteq{\tilde{Q} = M P^{-1} M^\top + D_{\sqrt{q}}\,\Xi\,D_{\sqrt{q}}}
      \end{ansatzbox}
      \fiteq{
        \Xi_{ii} = 1, \quad \Xi_{ij} = \langle \vec{g}, \vec{g}' \rangle
        \quad\text{with}\quad
        \vec{g}, \vec{g}' \sim \mathrm{Unif}\!\left(\mathbb{S}^{d-k-1}\right)}
      \adjustbox{max width=\linewidth}{%
        $\displaystyle \tilde{\mu}_p(t) = \frac{1}{p}\sum_{i=1}^p
          \delta_{\vec{m}_i(t),\, q_i(t)}$\; Flows following Saad\&Solla}
    \end{column}
  \end{columns}
  \note{The weights are following a field (given by the gradient of the risk).
    If I write the network as a functional of the weight distribution, then
    every particle is following the same field (this is the key of meanfield
    assumption), so I can simply write mass conservation on that field and I get
    the motion of the full measure.
    I would like to do the same with Saad\&Solla, but I can't directly.
    This ansatz turned out to be useful for other description we see in a
    moment.}
\end{frame}

% === 19. The information exponent ==========================================
\begin{frame}{The information exponent}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \vspace{1em}
      \begin{center}\large Guiding quantity for all our analysis\end{center}
      \vspace{1em}
      \begin{thmbox}{Theorem}
        The sample complexity is given by:
        \[
          n = \begin{cases}
            \mathcal{O}(d) & \ell = 1, \\
            \mathcal{O}(d\log d) & \ell = 2, \\
            \mathcal{O}(d^{\ell-1}) & \ell \geq 3.
          \end{cases}
        \]
      \end{thmbox}
    \end{column}
    \begin{column}{0.54\textwidth}
      \begin{defbox}{Information Exponent}
        \fiteq{
          \ell = \min\Bigl\{ j \geq 1 : a_j = \frac{1}{j!}\,
            \mathbb{E}_{z\sim\mathcal{N}(0,1)}[h^\star(z)\,\mathrm{He}_j(z)] \neq 0 \Bigr\}}
      \end{defbox}
      \vspace{1.5em}

      Examples:
      \begin{itemize}
        \item $h^\star = z^2$ has IE $= 2$
        \item $h^\star = \mathrm{He}_\ell(z)$ has IE $= \ell$
      \end{itemize}
      \vspace{1em}
      \begin{center}
        \aside{This is not theoretical minimum, you could use a partial trace
          algorithm to push it down to $d^{\ell/2}$}
      \end{center}
    \end{column}
  \end{columns}
  \note{We are implicitly assuming that the activation functions ``excites
    enough Hermite modes''.
    We study phase retrieval extensively in the previous slides.}
\end{frame}

\begin{frame}{Is the IE the actual lower bound in complexity?}

% === 22. How this work? ====================================================
\begin{frame}{How this work?}
  \begin{center}
    {\large One-pass SGD is linear in the label}\quad
    $\displaystyle \nabla_{\vec{w}}\mathcal{R}
      = -\,\mathbb{E}\left[y\,\sigma'(\lambda)\,\frac{\vec{x}}{\sqrt{d}}\right]$
  \end{center}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      What if we apply a generic transformation
      $\mathbb{E}\bigl[g(y)\,\varphi(\vec{x})\bigr]$?
      \begin{defbox}{Generative Exponent}
        \fiteq{
          \ell^\star = \min\Bigl\{ j \geq 1 :
            \mathbb{E}_{z\sim\mathcal{N}(0,1)}\bigl[g(h^\star(z))\,\mathrm{He}_j(z)\bigr]
            \neq 0 \textit{ for some } g : \mathbb{R} \to \mathbb{R} \Bigr\}}
      \end{defbox}
      \vspace{0.2em}

      \emph{Examples}:
      \begin{itemize}
        \footnotesize
        \item $\mathrm{He}_{2k+1}$ has IE $= 2k+1$, while GE $= 1$
        \item $\mathrm{He}_{2k}$ has IE $= 2$, while GE $= 2$
        \item $\mathrm{sign}(z_1 z_2 z_3)$ has IE $=$ GE $= 3$
      \end{itemize}
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering The double gradient induces a transformation
      \fiteq{
        \nabla_{\vec{w}}\ell\bigl(f(\vec{x};\tilde{\vec{w}}), y\bigr)
          = -\,y\,\sigma'(\tilde{\lambda})\,\frac{\vec{x}}{\sqrt{d}},
        \qquad
        \tilde{\lambda} = \frac{\langle\tilde{\vec{w}}, \vec{x}\rangle}{\sqrt{d}}}
      \fiteq{
        \nabla_{\vec{w}}\ell\bigl(f(\vec{x};\tilde{\vec{w}}), y\bigr)
          \approx -\,\frac{\vec{x}}{\sqrt{d}}
            \left(y\,\sigma'(\lambda) + \rho\,y^2\,\sigma'(\lambda)\sigma''(\lambda)
              + \frac{\rho^2}{2}\,y^3\,\sigma'(\lambda)^2\sigma'''(\lambda) + \cdots\right)}
      \vspace{-0.6em}
      \begin{thmbox}{Theorem (Repetita Iuvant)}
        \centering
        Sample complexity of EGD
        \vspace{-0.9em}
        \[
          \footnotesize
          n = \begin{cases}
            \mathcal{O}(d) & \ell^\star = 1, \\
            \mathcal{O}(d\log d) & \ell^\star = 2.
          \end{cases}
        \]
        \vspace{-0.4em}

        {\scriptsize\color{black!62} We do not have results for
          $\ell^\star > 2$, because we worked on polynomials}
      \end{thmbox}
    \end{column}
  \end{columns}
  \note{TODO rewrite the transformed gradient without $\varphi$.
    Clarify how we work only on polynomial in our current proof.
    (Not applied: the thesis writes the query as
    $\mathbb{E}[g(y)\varphi(\vec{x})]$ too, and both displayed gradients here
    already carry no $\varphi$ -- your call.)}
\end{frame}

% === 23. IE & Attention model ==============================================
\begin{frame}{IE \& Attention model}
  \adjustbox{max width=\linewidth}{%
    {\large Our proxy for attention:}\;
    $\displaystyle
      f(X) = \vec{a}_L^\top \SoftMax\!\left(\left(X + \frac{P}{\sqrt{d}}\right)
        \vec{w}\vec{w}^\top \left(X + \frac{P}{\sqrt{d}}\right)^\top\right)\vec{a}_R$
    \; $X \in \mathbb{R}^{L \times d}$}
  \vspace{0.2em}
  \begin{center}
    {\large Sufficient statistics:}\quad
    $\displaystyle m = \frac{\vec{w}_\star^\top\vec{w}}{d}
      \quad\text{and}\quad \vec{e} = \frac{P\vec{w}}{\sqrt{d}}$
  \end{center}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{defbox}{Sequence Information Exponent (SIE)}
        \fiteq{
          \mathit{SIE}(\vec{f}^{\,\mathrm{SSI}}_{\vec{w}_\star}) \coloneqq
            \min\left\{\sum_{l=1}^L k_l > 0 : \vec{k} \in \mathbb{N}^L,\;
              \mathbb{E}_{\vec{z}\sim\mathcal{N}(0,I_L)}\left[
                \left(\prod_{l=1}^L \mathrm{He}_{k_l}(z_l)\right) g(\vec{z})
              \right] \neq 0 \right\}}
      \end{defbox}
      \vspace{0.2em}

      \emph{Examples}:
      \begin{itemize}
        \scriptsize
        \item $g(\vec{z}_\star) = z_{\star,1}z_{\star,2}
          = \mathrm{He}_1(z_{\star,1})\mathrm{He}_1(z_{\star,2})$ has SIE $= 2$;
        \item $g(\vec{z}_\star) = \mathrm{He}_1(z_{\star,1})\mathrm{He}_4(z_{\star,2})
          + \mathrm{He}_2(z_{\star,3})\mathrm{He}_2(z_{\star,4})$ has SIE $= 4$;
        \item $g(\vec{z}_\star) = \prod_{l=1}^L \mathrm{He}_{k_l}(z_{\star,l})$
          has SIE $= \sum_{l=1}^L k_l$.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{thmbox}{Theorem (Repetita Iuvant)}
        \centering
        Sample complexity of EGD
        \vspace{-0.8em}
        \[
          \small
          \begin{cases}
            \mathcal{O}_L(d) & \textit{if } \mathit{SIE} = 1 \\
            \mathcal{O}_L(d\log d) & \textit{if } \mathit{SIE} = 2 \\
            \mathcal{O}_L(d^{\mathit{SIE}-1}) & \textit{if } \mathit{SIE} \geq 3
          \end{cases}
        \]
      \end{thmbox}
    \end{column}
  \end{columns}
  \note{Now the data is a set of $L$ tokens of dimension $d$. Our attantion
    model is very simplified: tied QK, V is identity, no heads.
    Who is using neural networks this days? We are all into transformers!
    This is our proxy.}
\end{frame}
